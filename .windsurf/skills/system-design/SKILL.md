export default function VibePlanner() {
  const [prompt, setPrompt] = useState("");
  const [loading, setLoading] = useState(false);
  const [activePhase, setActivePhase] = useState(null);
  const [plan, setPlan] = useState(null);
  const [currentPhaseIdx, setCurrentPhaseIdx] = useState(-1);
  const [error, setError] = useState(null);
  const textareaRef = useRef(null);

  const generatePlan = async () => {
    if (!prompt.trim()) return;
    setLoading(true);
    setError(null);
    setPlan(null);
    setActivePhase(null);
    setCurrentPhaseIdx(-1);

    // Animate phases one by one
    for (let i = 0; i < PHASES.length; i++) {
      setCurrentPhaseIdx(i);
      await new Promise(r => setTimeout(r, 600));
    }

    try {
      const response = await fetch("https://api.anthropic.com/v1/messages", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          model: "claude-sonnet-4-20250514",
          max_tokens: 4000,
          system: SYSTEM_PROMPT,
          messages: [{ role: "user", content: `Project idea: ${prompt}` }],
        }),
      });

      const data = await response.json();
      const text = data.content?.map(b => b.text || "").join("");
      const clean = text.replace(/```json|```/g, "").trim();
      const parsed = JSON.parse(clean);
      setPlan(parsed);
      setActivePhase("requirements");
    } catch (e) {
      setError("কিছু একটা সমস্যা হয়েছে। আবার চেষ্টা করুন।");
    } finally {
      setLoading(false);
      setCurrentPhaseIdx(-1);
    }
  };

  const renderPhaseContent = (phase) => {
    if (!plan || !plan[phase]) return null;
    const data = plan[phase];

    switch (phase) {
      case "requirements":
        return (
          <div style={styles.phaseContent}>
            <Section title="📌 Functional Requirements" items={data.functional} color="#6366f1" />
            <Section title="⚡ Non-Functional Requirements" items={data.nonFunctional} color="#8b5cf6" />
            <Section title="👤 User Stories" items={data.userStories} color="#a855f7" />
          </div>
        );
      case "architecture":
        return (
          <div style={styles.phaseContent}>
            <div style={styles.overviewBox}>{data.overview}</div>
            <div style={styles.techGrid}>
              {Object.entries(data.techStack || {}).map(([key, val]) => (
                <div key={key} style={styles.techCard}>
                  <div style={styles.techLabel}>{key.toUpperCase()}</div>
                  <div style={styles.techVal}>{val}</div>
                </div>
              ))}
            </div>
            <Section title="🧩 Components" items={data.components} color="#6366f1" />
          </div>
        );
      case "database":
        return (
          <div style={styles.phaseContent}>
            {(data.tables || []).map((table, i) => (
              <div key={i} style={styles.tableCard}>
                <div style={styles.tableName}>🗂 {table.name}</div>
                <div style={styles.tablePurpose}>{table.purpose}</div>
                <div style={styles.fieldList}>
                  {(table.fields || []).map((f, j) => (
                    <div key={j} style={styles.field}>• {f}</div>
                  ))}
                </div>
              </div>
            ))}
          </div>
        );
      case "api":
        return (
          <div style={styles.phaseContent}>
            {(data.endpoints || []).map((ep, i) => (
              <div key={i} style={styles.endpoint}>
                <span style={{ ...styles.method, background: methodColor(ep.method) }}>{ep.method}</span>
                <span style={styles.path}>{ep.path}</span>
                {ep.auth && <span style={styles.authBadge}>🔒 Auth</span>}
                <div style={styles.epDesc}>{ep.description}</div>
              </div>
            ))}
          </div>
        );
      case "sprints":
        return (
          <div style={styles.phaseContent}>
            {(data.sprints || []).map((s, i) => (
              <div key={i} style={styles.sprintCard}>
                <div style={styles.sprintHeader}>
                  <span style={styles.sprintNum}>Sprint {s.sprint}</span>
                  <span style={styles.sprintDuration}>{s.duration}</span>
                </div>
                <div style={styles.sprintGoal}>🎯 {s.goal}</div>
                {(s.tasks || []).map((t, j) => (
                  <div key={j} style={styles.task}>✓ {t}</div>
                ))}
              </div>
            ))}
          </div>
        );
      case "security":
        return (
          <div style={styles.phaseContent}>
            <Section title="🔐 Security Measures" items={data.security} color="#f43f5e" />
            <Section title="📈 Scalability" items={data.scalability} color="#f97316" />
            <Section title="📊 Monitoring" items={data.monitoring} color="#eab308" />
          </div>
        );
      default: return null;
    }
  };

  return (
    <div style={styles.root}>
      {/* Header */}
      <div style={styles.header}>
        <div style={styles.logo}>⚡ VibePlanner</div>
        <div style={styles.subtitle}>Simple prompt → Enterprise-grade development plan</div>
      </div>

      {/* Input */}
      <div style={styles.inputSection}>
        <textarea
          ref={textareaRef}
          value={prompt}
          onChange={e => setPrompt(e.target.value)}
          placeholder="আপনার project idea লিখুন... (e.g., Build an ecommerce website)"
          style={styles.textarea}
          rows={3}
          onKeyDown={e => { if (e.key === "Enter" && e.ctrlKey) generatePlan(); }}
        />
        <button
          onClick={generatePlan}
          disabled={loading || !prompt.trim()}
          style={{ ...styles.btn, opacity: (loading || !prompt.trim()) ? 0.5 : 1 }}
        >
          {loading ? "⚙️ Generating..." : "🚀 Generate Plan"}
        </button>
        <div style={styles.hint}>Ctrl+Enter to generate</div>
      </div>

      {/* Loading */}
      {loading && (
        <div style={styles.loadingSection}>
          {PHASES.map((p, i) => (
            <div key={p.id} style={{
              ...styles.loadPhase,
              opacity: i <= currentPhaseIdx ? 1 : 0.3,
              transform: i === currentPhaseIdx ? "scale(1.05)" : "scale(1)",
            }}>
              <span>{p.icon}</span>
              <span style={{ color: i <= currentPhaseIdx ? p.color : "#555" }}>{p.label}</span>
              {i < currentPhaseIdx && <span style={{ color: "#4ade80" }}>✓</span>}
              {i === currentPhaseIdx && <span style={styles.spinner}>⟳</span>}
            </div>
          ))}
        </div>
      )}

      {/* Error */}
      {error && <div style={styles.error}>{error}</div>}

      {/* Plan */}
      {plan && (
        <div style={styles.planSection}>
          {/* Phase Tabs */}
          <div style={styles.tabs}>
            {PHASES.map(p => (
              <button
                key={p.id}
                onClick={() => setActivePhase(p.id)}
                style={{
                  ...styles.tab,
                  background: activePhase === p.id ? p.color : "transparent",
                  borderColor: p.color,
                  color: activePhase === p.id ? "#fff" : p.color,
                }}
              >
                {p.icon} {p.label}
              </button>
            ))}
          </div>

          {/* Phase Detail */}
          <div style={styles.detail}>
            {activePhase && plan[activePhase] && (
              <>
                <h2 style={styles.phaseTitle}>{plan[activePhase].title}</h2>
                {renderPhaseContent(activePhase)}
              </>
            )}
          </div>
        </div>
      )}
    </div>
  );
}

function Section({ title, items = [], color }) {
  return (
    <div style={{ marginBottom: 20 }}>
      <div style={{ color, fontWeight: 700, marginBottom: 8, fontSize: 15 }}>{title}</div>
      {items.map((item, i) => (
        <div key={i} style={{ padding: "6px 10px", marginBottom: 5, background: "#1a1a2e", borderLeft: `3px solid ${color}`, borderRadius: 4, fontSize: 14, color: "#cbd5e1" }}>
          {item}
        </div>
      ))}
    </div>
  );
}

function methodColor(method) {
  return { GET: "#22c55e", POST: "#3b82f6", PUT: "#f59e0b", DELETE: "#ef4444", PATCH: "#8b5cf6" }[method] || "#6b7280";
}

const getIcon = (name, type) => {
  if (type === "root") return "◈";
  if (type === "file") {
    if (name.endsWith(".py")) return "🐍";
    if (name.endsWith(".ts") || name.endsWith(".tsx")) return "⬡";
    if (name.endsWith(".yml") || name.endsWith(".yaml")) return "⚙";
    if (name.endsWith(".json")) return "{}";
    if (name.endsWith(".sh")) return "▶";
    if (name === "Dockerfile") return "🐳";
    return "◻";
  }
  return "◫";
};

function TreeNode({ node, depth = 0, defaultOpen = false }) {
  const [open, setOpen] = useState(defaultOpen || depth < 1);
  const [hovered, setHovered] = useState(false);
  const isDir = node.type === "dir" || node.type === "root";
  const hasChildren = node.children && node.children.length > 0;
  const color = node.color || (depth === 0 ? "#60a5fa" : depth === 1 ? "#34d399" : "#94a3b8");
  const indent = depth * 20;

  return (
    <div>
      <div
        onClick={() => isDir && hasChildren && setOpen(o => !o)}
        onMouseEnter={() => setHovered(true)}
        onMouseLeave={() => setHovered(false)}
        style={{
          display: "flex",
          alignItems: "flex-start",
          gap: 10,
          padding: "5px 12px 5px",
          paddingLeft: 12 + indent,
          cursor: isDir && hasChildren ? "pointer" : "default",
          background: hovered ? "rgba(255,255,255,0.04)" : "transparent",
          borderRadius: 6,
          transition: "background 0.15s",
          borderLeft: depth > 0 ? `1px solid rgba(255,255,255,0.06)` : "none",
          marginLeft: depth > 0 ? 8 : 0,
        }}
      >
        {/* Toggle arrow */}
        <span style={{ width: 14, flexShrink: 0, color: "#4B5563", fontSize: 10, marginTop: 3 }}>
          {isDir && hasChildren ? (open ? "▼" : "▶") : ""}
        </span>

        {/* Icon */}
        <span style={{ fontSize: 13, flexShrink: 0, marginTop: 1, opacity: 0.9 }}>
          {getIcon(node.name, node.type)}
        </span>

        {/* Name */}
        <div style={{ flex: 1, minWidth: 0 }}>
          <span style={{
            fontFamily: "'JetBrains Mono', 'Fira Code', monospace",
            fontSize: 13,
            fontWeight: isDir ? 600 : 400,
            color: isDir ? color : "#CBD5E1",
            letterSpacing: isDir ? 0.3 : 0,
          }}>
            {node.name}
          </span>
          {node.desc && (
            <span style={{
              marginLeft: 12,
              fontSize: 11,
              color: "#475569",
              fontFamily: "system-ui, sans-serif",
            }}>
              — {node.desc}
            </span>
          )}
        </div>
      </div>

      {isDir && hasChildren && open && (
        <div>
          {node.children.map((child, i) => (
            <TreeNode key={i} node={child} depth={depth + 1} />
          ))}
        </div>
      )}
    </div>
  );
}

    const lines = [`CREATE TABLE ${table.name} (`];
    table.fields.forEach((f, i) => {
      const constraints = [
        f.pk ? "PRIMARY KEY DEFAULT gen_random_uuid()" : "",
        f.nn && !f.pk ? "NOT NULL" : "",
        f.uk ? "UNIQUE" : "",
        f.fk ? `REFERENCES ${f.fk} ON DELETE CASCADE` : "",
      ].filter(Boolean).join(" ");
      lines.push(`  ${f.name.padEnd(28)} ${f.type.padEnd(20)} ${constraints}${i < table.fields.length - 1 ? "," : ""}`);
    });

    const idxFields = table.fields.filter(f => f.idx && !f.pk);
    idxFields.forEach(f => {
